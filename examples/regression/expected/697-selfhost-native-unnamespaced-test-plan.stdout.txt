plan = 0,1
import sys.process as process

main {
    0 => failures!
    "running 1 tests" -> println
    test_expected_failure() -> if {
        "test test_expected_failure ... ok" -> println
    } else {
        "test test_expected_failure ... FAILED" -> println
        failures! + 1 => failures!
    }
    "test result: $(1 - failures!) passed; $(failures!) failed" -> println
    failures! == 0 -> if { 0 -> process.exit } else { 1 -> process.exit }
}
