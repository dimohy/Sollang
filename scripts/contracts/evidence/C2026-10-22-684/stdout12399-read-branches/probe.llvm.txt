@sollang_runtime_newline = private constant [1 x i8] c"\0A"
@sollang_runtime_io_failed = internal global i8 0

define internal void @sollang_runtime_flush_stdout() {
entry:
  ret void
}
define internal i1 @sollang_runtime_write_stdout(ptr %data, i64 %len) {
entry:
  %empty = icmp eq i64 %len, 0
  br i1 %empty, label %success, label %write
write:
  %written = call i64 @write(i32 1, ptr %data, i64 %len)
  %complete = icmp eq i64 %written, %len
  br i1 %complete, label %success, label %failure
failure:
  store atomic i8 1, ptr @sollang_runtime_io_failed monotonic, align 1
  ret i1 false
success:
  ret i1 true
}
define internal i1 @sollang_runtime_append_stdout_checked(ptr %data, i64 %len) {
entry:
  %ok = call i1 @sollang_runtime_write_stdout(ptr %data, i64 %len)
  ret i1 %ok
}
define internal i1 @sollang_runtime_print_checked(ptr %data, i64 %len, i1 %newline) {
entry:
  %data_ok = call i1 @sollang_runtime_append_stdout_checked(ptr %data, i64 %len)
  br i1 %newline, label %write_newline, label %done
write_newline:
  %newline_ok = call i1 @sollang_runtime_append_stdout_checked(ptr @sollang_runtime_newline, i64 1)
  %combined = and i1 %data_ok, %newline_ok
  br label %done
done:
  %ok = phi i1 [ %data_ok, %entry ], [ %combined, %write_newline ]
  ret i1 %ok
}
define internal void @sollang_runtime_print(ptr %data, i64 %len, i1 %newline) {
entry:
  %ok = call i1 @sollang_runtime_print_checked(ptr %data, i64 %len, i1 %newline)
  ret void
}
define internal i1 @sollang_runtime_write_stderr(ptr %data, i64 %len) {
entry:
  %empty = icmp eq i64 %len, 0
  br i1 %empty, label %success, label %write
write:
  %offset = phi i64 [ 0, %entry ], [ %next, %advance ]
  %remaining = sub i64 %len, %offset
  %current = getelementptr i8, ptr %data, i64 %offset
  %count = call i64 @write(i32 2, ptr %current, i64 %remaining)
  %positive = icmp sgt i64 %count, 0
  br i1 %positive, label %advance, label %failure
advance:
  %next = add i64 %offset, %count
  %complete = icmp eq i64 %next, %len
  br i1 %complete, label %success, label %write
failure:
  store atomic i8 1, ptr @sollang_runtime_io_failed monotonic, align 1
  ret i1 false
success:
  ret i1 true
}
define internal void @sollang_runtime_eprint(ptr %data, i64 %len, i1 %newline) {
entry:
  %ok = call i1 @sollang_runtime_write_stderr(ptr %data, i64 %len)
  br i1 %newline, label %write_newline, label %done
write_newline:
  %newline_ok = call i1 @sollang_runtime_write_stderr(ptr @sollang_runtime_newline, i64 1)
  br label %done
done:
  ret void
}
define internal void @sollang_runtime_eprintln(ptr %data, i64 %len) {
entry:
  call void @sollang_runtime_eprint(ptr %data, i64 %len, i1 true)
  ret void
}
define i32 @sollang_m131_s2(%sollang.text %arg) {
entry:
  %data = extractvalue %sollang.text %arg, 0
  %length = extractvalue %sollang.text %arg, 1
  %prompt_ok = call i1 @sollang_runtime_print_checked(ptr %data, i64 %length, i1 false)
  %value_slot = alloca i32, align 4
  %started_slot = alloca i1, align 1
  %negative_slot = alloca i1, align 1
  store i32 0, ptr %value_slot, align 4
  store i1 false, ptr %started_slot, align 1
  store i1 false, ptr %negative_slot, align 1
  br i1 %prompt_ok, label %read, label %done
read:
  %character = call i32 @getchar()
  %eof = icmp eq i32 %character, -1
  br i1 %eof, label %done, label %classify
classify:
  %digit_low = icmp sge i32 %character, 48
  %digit_high = icmp sle i32 %character, 57
  %digit = and i1 %digit_low, %digit_high
  br i1 %digit, label %append, label %non_digit
append:
  %previous = load i32, ptr %value_slot, align 4
  %scaled = mul i32 %previous, 10
  %digit_value = sub i32 %character, 48
  %updated = add i32 %scaled, %digit_value
  store i32 %updated, ptr %value_slot, align 4
  store i1 true, ptr %started_slot, align 1
  br label %read
non_digit:
  %started = load i1, ptr %started_slot, align 1
  br i1 %started, label %done, label %prefix
prefix:
  %minus = icmp eq i32 %character, 45
  br i1 %minus, label %mark_negative, label %read
mark_negative:
  store i1 true, ptr %negative_slot, align 1
  br label %read
done:
  %magnitude = load i32, ptr %value_slot, align 4
  %negative = load i1, ptr %negative_slot, align 1
  %negated = sub i32 0, %magnitude
  %result = select i1 %negative, i32 %negated, i32 %magnitude
  ret i32 %result
}

%sollang.text = type { ptr, i64 }
@probe_mode = global i1 true
@probe_reads = global i32 0
define i64 @write(i32 %fd, ptr %data, i64 %len) {
 %fail = load i1, ptr @probe_mode
 %count = select i1 %fail, i64 -1, i64 %len
 ret i64 %count
}
define i32 @getchar() {
 %prior = load i32, ptr @probe_reads
 %next = add i32 %prior, 1
 store i32 %next, ptr @probe_reads
 %first = icmp eq i32 %prior, 0
 %second = icmp eq i32 %prior, 1
 %tail = select i1 %second, i32 50, i32 10
 %character = select i1 %first, i32 52, i32 %tail
 ret i32 %character
}
define i32 @main() {
 %prompt = insertvalue %sollang.text zeroinitializer, i64 1, 1
 %failed_result = call i32 @sollang_m131_s2(%sollang.text %prompt)
 %failed_reads = load i32, ptr @probe_reads
 %not_read = icmp eq i32 %failed_reads, 0
 %zero = icmp eq i32 %failed_result, 0
 %failure_state = load atomic i8, ptr @sollang_runtime_io_failed monotonic, align 1
 %failed = icmp eq i8 %failure_state, 1
 %failed_a = and i1 %not_read, %zero
 %failed_pass = and i1 %failed_a, %failed
 store i1 false, ptr @probe_mode
 %healthy_result = call i32 @sollang_m131_s2(%sollang.text %prompt)
 %healthy_reads = load i32, ptr @probe_reads
 %read_three = icmp eq i32 %healthy_reads, 3
 %forty_two = icmp eq i32 %healthy_result, 42
 %retained_state = load atomic i8, ptr @sollang_runtime_io_failed monotonic, align 1
 %retained = icmp eq i8 %retained_state, 1
 %healthy_a = and i1 %read_three, %forty_two
 %healthy_pass = and i1 %healthy_a, %retained
 %passed = and i1 %failed_pass, %healthy_pass
 %exit = select i1 %passed, i32 0, i32 1
 ret i32 %exit
}
