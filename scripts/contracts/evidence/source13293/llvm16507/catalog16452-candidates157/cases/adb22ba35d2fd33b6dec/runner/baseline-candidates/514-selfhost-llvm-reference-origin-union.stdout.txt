define internal i32 @sollang_directory_compare_nodes(ptr %left, ptr %right) {

  br i1 %before, label %insert, label %advance
advance:
  %current_next_slot = getelementptr %sollang.directory_node, ptr %current, i32 0, i32 0
  %next = load ptr, ptr %current_next_slot, align 8
  br label %scan
insert:
  %node_next = getelementptr %sollang.directory_node, ptr %node, i32 0, i32 0
  store ptr %current, ptr %node_next, align 8
  %previous_next = getelementptr %sollang.directory_node, ptr %previous, i32 0, i32 0
  store ptr %node, ptr %previous_next, align 8
  ret ptr %head
append:
  %append_node_next = getelementptr %sollang.directory_node, ptr %node, i32 0, i32 0
  store ptr null, ptr %append_node_next, align 8
  %append_previous_next = getelementptr %sollang.directory_node, ptr %previous, i32 0, i32 0
  store ptr %node, ptr %append_previous_next, align 8
  ret ptr %head
}

define internal void @sollang_directory_free_nodes(ptr %head) {
entry:
  br label %loop
loop:
  %node = phi ptr [ %head, %entry ], [ %next, %body ]
  %done = icmp eq ptr %node, null
  br i1 %done, label %finish, label %body
body:
  %next_slot = getelementptr %sollang.directory_node, ptr %node, i32 0, i32 0
  %next = load ptr, ptr %next_slot, align 8
  call void @free(ptr %node)
  br label %loop
finish:
  ret void
}

define internal ptr @sollang_directory_serialize(ptr %head, i64 %total) {
entry:
  %empty = icmp eq i64 %total, 0
  br i1 %empty, label %empty_result, label %allocate
empty_result:
  ret ptr null
allocate:
  %output = call ptr @malloc(i64 %total)
  %allocated = icmp ne ptr %output, null
  br i1 %allocated, label %loop, label %allocation_failed
allocation_failed:
  call void @sollang_directory_free_nodes(ptr %head)
  ret ptr null
loop:
  %node = phi ptr [ %head, %allocate ], [ %next_node, %body ]
  %offset = phi i64 [ 0, %allocate ], [ %next_offset, %body ]
  %done = icmp eq ptr %node, null
  br i1 %done, label %finish, label %body
body:
  %next_slot = getelementptr %sollang.directory_node, ptr %node, i32 0, i32 0
  %next_node = load ptr, ptr %next_slot, align 8
  %length_slot = getelementptr %sollang.directory_node, ptr %node, i32 0, i32 1
  %length = load i64, ptr %length_slot, align 8
  %kind_slot = getelementptr %sollang.directory_node, ptr %node, i32 0, i32 2
  %kind = load i8, ptr %kind_slot, align 1
  %record = getelementptr i8, ptr %output, i64 %offset
  store i8 %kind, ptr %record, align 1
  %length32 = trunc i64 %length to i32
  %length0 = trunc i32 %length32 to i8
  %length_shift1 = lshr i32 %length32, 8
  %length1 = trunc i32 %length_shift1 to i8
  %length_shift2 = lshr i32 %length32, 16
  %length2 = trunc i32 %length_shift2 to i8
  %length_shift3 = lshr i32 %length32, 24
  %length3 = trunc i32 %length_shift3 to i8
  %length_ptr0 = getelementptr i8, ptr %record, i64 1
  %length_ptr1 = getelementptr i8, ptr %record, i64 2
  %length_ptr2 = getelementptr i8, ptr %record, i64 3
  %length_ptr3 = getelementptr i8, ptr %record, i64 4
  store i8 %length0, ptr %length_ptr0, align 1
  store i8 %length1, ptr %length_ptr1, align 1
  store i8 %length2, ptr %length_ptr2, align 1
  store i8 %length3, ptr %length_ptr3, align 1
  %destination = getelementptr i8, ptr %record, i64 5
  %source = getelementptr i8, ptr %node, i64 24
  call void @sollang_directory_copy(ptr %destination, ptr %source, i64 %length)
  %record_size = add i64 %length, 5
  %next_offset = add i64 %offset, %record_size
  call void @free(ptr %node)
  br label %loop
finish:
  ret ptr %output
}

define internal %sollang.path_absolute_result @sollang_platform_absolute_path(ptr %path, i64 %len, i32 %style) {
entry:
  %native_style = icmp eq i32 %style, 0
  %nonempty = icmp ugt i64 %len, 0
  %valid = and i1 %native_style, %nonempty
  br i1 %valid, label %scan, label %fail
scan:
  %scan_index = phi i64 [0, %entry], [%scan_next, %scan_step]
  %byte_ptr = getelementptr i8, ptr %path, i64 %scan_index
  %byte = load i8, ptr %byte_ptr, align 1
  %nul = icmp eq i8 %byte, 0
  br i1 %nul, label %fail, label %scan_step
scan_step:
  %scan_next = add i64 %scan_index, 1
  %scan_done = icmp eq i64 %scan_next, %len
  br i1 %scan_done, label %select_base, label %scan
select_base:
  %first = load i8, ptr %path, align 1
  %absolute_input = icmp eq i8 %first, 47
  br i1 %absolute_input, label %without_cwd, label %with_cwd
with_cwd:
  %cwd = call ptr @getcwd(ptr null, i64 0)
  %cwd_ok = icmp ne ptr %cwd, null
  br i1 %cwd_ok, label %measure_cwd, label %fail
measure_cwd:
  %cwd_len = call i64 @strlen(ptr %cwd)
  %prefix_len = add i64 %cwd_len, 1
  br label %allocate_input
without_cwd:
  br label %allocate_input
allocate_input:
  %base = phi ptr [%cwd, %measure_cwd], [null, %without_cwd]
  %base_len = phi i64 [%cwd_len, %measure_cwd], [0, %without_cwd]
  %prefix = phi i64 [%prefix_len, %measure_cwd], [0, %without_cwd]
  %total = add i64 %prefix, %len
  %capacity = add i64 %total, 1
  %sum_ok = icmp uge i64 %total, %len
  %capacity_ok = icmp ugt i64 %capacity, %total
  %allocation_size_ok = and i1 %sum_ok, %capacity_ok
  br i1 %allocation_size_ok, label %allocate, label %base_fail
allocate:
  %combined = call ptr @malloc(i64 %capacity)
  %combined_ok = icmp ne ptr %combined, null
  br i1 %combined_ok, label %copy_base, label %base_fail
copy_base:
  br i1 %absolute_input, label %copy_path, label %copy_cwd
copy_cwd:
  call void @llvm.memcpy.p0.p0.i64(ptr %combined, ptr %base, i64 %base_len, i1 false)
  %base_end = getelementptr i8, ptr %combined, i64 %base_len
  store i8 47, ptr %base_end, align 1
  br label %copy_path
copy_path:
  %destination = getelementptr i8, ptr %combined, i64 %prefix
  call void @llvm.memcpy.p0.p0.i64(ptr %destination, ptr %path, i64 %len, i1 false)
  call void @free(ptr %base)
  %output = call ptr @malloc(i64 %capacity)
  %output_ok = icmp ne ptr %output, null
  br i1 %output_ok, label %initialize, label %combined_fail
initialize:
  store i8 47, ptr %output, align 1
  br label %components
components:
  %cursor = phi i64 [0, %initialize], [%separator_next, %separator], [%component_end, %publish]
  %output_len = phi i64 [1, %initialize], [%output_len, %separator], [%published_len, %publish]
  %done = icmp eq i64 %cursor, %total
  br i1 %done, label %complete, label %inspect
inspect:
  %component_ptr = getelementptr i8, ptr %combined, i64 %cursor
  %component_first = load i8, ptr %component_ptr, align 1
  %is_separator = icmp eq i8 %component_first, 47
  br i1 %is_separator, label %separator, label %find_end
separator:
  %separator_next = add i64 %cursor, 1
  br label %components
find_end:
  %end_cursor = phi i64 [%cursor, %inspect], [%end_next, %end_step]
  %at_end = icmp eq i64 %end_cursor, %total
  br i1 %at_end, label %classify, label %end_byte
end_byte:
  %end_ptr = getelementptr i8, ptr %combined, i64 %end_cursor
  %end_value = load i8, ptr %end_ptr, align 1
  %at_separator = icmp eq i8 %end_value, 47
  br i1 %at_separator, label %classify, label %end_step
end_step:
  %end_next = add i64 %end_cursor, 1
  br label %find_end
classify:
  %component_end = phi i64 [%end_cursor, %find_end], [%end_cursor, %end_byte]
  %component_len = sub i64 %component_end, %cursor
  %first_dot = icmp eq i8 %component_first, 46
  %one_byte = icmp eq i64 %component_len, 1
  %dot = and i1 %first_dot, %one_byte
  br i1 %dot, label %unchanged, label %check_parent
check_parent:
  %two_bytes = icmp eq i64 %component_len, 2
  %possible_parent = and i1 %first_dot, %two_bytes
  br i1 %possible_parent, label %parent_second, label %append
parent_second:
  %second_ptr = getelementptr i8, ptr %component_ptr, i64 1
  %second = load i8, ptr %second_ptr, align 1
  %parent = icmp eq i8 %second, 46
  br i1 %parent, label %pop, label %append
pop:
  %pop_index = phi i64 [%output_len, %parent_second], [%previous, %pop_byte]
  %at_root = icmp eq i64 %pop_index, 1
  br i1 %at_root, label %popped_root, label %pop_byte
pop_byte:
  %previous = sub i64 %pop_index, 1
  %previous_ptr = getelementptr i8, ptr %output, i64 %previous
  %previous_byte = load i8, ptr %previous_ptr, align 1
  %previous_separator = icmp eq i8 %previous_byte, 47
  br i1 %previous_separator, label %popped, label %pop
popped:
  %is_root_separator = icmp eq i64 %previous, 0
  %popped_len = select i1 %is_root_separator, i64 1, i64 %previous
  br label %publish
popped_root:
  br label %publish
unchanged:
  br label %publish
append:
  %needs_separator = icmp ugt i64 %output_len, 1
  br i1 %needs_separator, label %append_separator, label %append_bytes
append_separator:
  %separator_ptr = getelementptr i8, ptr %output, i64 %output_len
  store i8 47, ptr %separator_ptr, align 1
  %after_separator = add i64 %output_len, 1
  br label %append_bytes
append_bytes:
  %append_offset = phi i64 [%output_len, %append], [%after_separator, %append_separator]
  %append_ptr = getelementptr i8, ptr %output, i64 %append_offset
  call void @llvm.memcpy.p0.p0.i64(ptr %append_ptr, ptr %component_ptr, i64 %component_len, i1 false)
  %appended_len = add i64 %append_offset, %component_len
  br label %publish
publish:
  %published_len = phi i64 [%popped_len, %popped], [1, %popped_root], [%output_len, %unchanged], [%appended_len, %append_bytes]
  br label %components
complete:
  %last_index = sub i64 %total, 1
  %last_ptr = getelementptr i8, ptr %combined, i64 %last_index
  %last = load i8, ptr %last_ptr, align 1
  %trailing_separator = icmp eq i8 %last, 47
  %nonroot_result = icmp ugt i64 %output_len, 1
  %retain_trailing = and i1 %trailing_separator, %nonroot_result
  br i1 %retain_trailing, label %trailing, label %success
trailing:
  %trailing_ptr = getelementptr i8, ptr %output, i64 %output_len
  store i8 47, ptr %trailing_ptr, align 1
  %trailing_len = add i64 %output_len, 1
  br label %success
success:
  %result_len = phi i64 [%output_len, %complete], [%trailing_len, %trailing]
  call void @free(ptr %combined)
  %result0 = insertvalue %sollang.path_absolute_result poison, ptr %output, 0
  %result1 = insertvalue %sollang.path_absolute_result %result0, i64 %result_len, 1
  %result2 = insertvalue %sollang.path_absolute_result %result1, i32 1, 2
  ret %sollang.path_absolute_result %result2
combined_fail:
  call void @free(ptr %combined)
  br label %fail
base_fail:
  call void @free(ptr %base)
  br label %fail
fail:
  ret %sollang.path_absolute_result zeroinitializer
}

define internal %sollang.directory_result @sollang_platform_read_directory(ptr %path, i64 %len, i32 %style) {
entry:

  br i1 %style_ok, label %prepare, label %fail
prepare:
  %buffer_size = add i64 %len, 1
  %path_buffer = call ptr @malloc(i64 %buffer_size)
  %path_ok = icmp ne ptr %path_buffer, null
  br i1 %path_ok, label %copy_path, label %fail
copy_path:
  call void @sollang_directory_copy(ptr %path_buffer, ptr %path, i64 %len)
  %zero_ptr = getelementptr i8, ptr %path_buffer, i64 %len
  store i8 0, ptr %zero_ptr, align 1

  call void @free(ptr %path_buffer)
  %opened = icmp ne ptr %directory, null
  br i1 %opened, label %enumerate, label %fail

  %head = phi ptr [ null, %copy_path ], [ %advanced_head, %advance ]
  %count = phi i64 [ 0, %copy_path ], [ %advanced_count, %advance ]
  %total = phi i64 [ 0, %copy_path ], [ %advanced_total, %advance ]
  %errno_ptr = call ptr @__errno_location()
  store i32 0, ptr %errno_ptr, align 4
  %entry_value = call ptr @readdir(ptr %directory)
  %at_end = icmp eq ptr %entry_value, null
  br i1 %at_end, label %finish_enumeration, label %scan_entry

name_length:

  %name_byte = load i8, ptr %name_slot, align 1

name_continue:

inspect_name:
  %first = load i8, ptr %name, align 1
  %second_ptr = getelementptr i8, ptr %name, i64 1
  %second = load i8, ptr %second_ptr, align 1
  %length_one = icmp eq i64 %name_index, 1
  %length_two = icmp eq i64 %name_index, 2
  %first_dot = icmp eq i8 %first, 46
  %second_dot = icmp eq i8 %second, 46
  %dot = and i1 %length_one, %first_dot
  %two_dots = and i1 %first_dot, %second_dot
  %dotdot = and i1 %length_two, %two_dots
  %special = or i1 %dot, %dotdot
  br i1 %special, label %skip, label %allocate_node
allocate_node:
  %node_size = add i64 %name_index, 24

initialize_node:

  store ptr null, ptr %node_next, align 8
  %node_length = getelementptr %sollang.directory_node, ptr %node, i32 0, i32 1
  store i64 %name_index, ptr %node_length, align 8
  %type_ptr = getelementptr i8, ptr %entry_value, i64 18
  %entry_type = load i8, ptr %type_ptr, align 1
  %is_symlink = icmp eq i8 %entry_type, 10
  %is_directory = icmp eq i8 %entry_type, 4
  %is_file = icmp eq i8 %entry_type, 8
  %file_or_other = select i1 %is_file, i8 0, i8 3
  %directory_or_other = select i1 %is_directory, i8 1, i8 %file_or_other

  %node_kind = getelementptr %sollang.directory_node, ptr %node, i32 0, i32 2
  store i8 %kind, ptr %node_kind, align 1
  %node_name = getelementptr i8, ptr %node, i64 24
  call void @sollang_directory_copy(ptr %node_name, ptr %name, i64 %name_index)
  %inserted_head = call ptr @sollang_directory_insert_sorted(ptr %head, ptr %node)
  %inserted_count = add i64 %count, 1
  %record_size = add i64 %name_index, 5
  %inserted_total = add i64 %total, %record_size

skip:

advance:

  %advanced_count = phi i64 [ %count, %skip ], [ %inserted_count, %initialize_node ]
