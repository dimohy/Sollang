namespace probes.suspension_source_control

struct Pair {
    left: Int
    right: Int
}

intChild: -> async Int {
    yield
    7
}

boolChild: -> async Bool {
    yield
    true
}

arrayChild: -> async [Int; ~] {
    yield
    [7, 2; ~]
}

pairChild: -> async Pair {
    yield
    Pair { left: 7, right: 2 }
}

increment value: Int -> Int => value + 1
sum left: Int, right: Int -> Int => left + right


enum Choice {
    First
    Second
}

choiceChild: -> async Choice {
    yield
    Choice.First
}

resultChild: -> async Result<Int, Text> {
    yield
    Result<Int, Text>.Ok(7)
}

audit value: Int, extra: Int -> Unit {}
consume value: Int -> Unit {}

case_subject_when: -> async Int {
    intChild() => pending
    pending -> await -> when {
        7 => 1
        else => 0
    }
}

main {}
