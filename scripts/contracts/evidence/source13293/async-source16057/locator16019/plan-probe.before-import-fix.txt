import sollang.compiler.semantic.context as semanticContext
import sollang.compiler.ir.typed as typedIr
import sollang.compiler.syntax.diagnostics as syntaxDiagnostics
import sollang.compiler.syntax.source_decoding as sourceDecoding
import sys.file as file
import sys.process as process

main {
    process.arguments() => arguments
    arguments[1] -> file.mapText => owner
    owner -> sourceDecoding.utf8Text => source
    [source,; ~] -> semanticContext.prepare => prepared
    prepared -> typedIr.coroutinePlanContext => plan
    plan.points -> each point {
        prepared.package.ranges[point.sourceModule] => range
        plan.ir[point.functionIr] => function
        prepared.package.symbols[range.symbolStart + function.symbol] => symbol
        prepared.package.tokens[range.tokenStart + symbol.nameToken] => name
        prepared.package.nodes[range.astStart + point.awaitAst] => site
        prepared.package.tokens[range.tokenStart + site.payloadToken] => token
        source -> syntaxDiagnostics.sourceLocation(token.span.start) => location
        source -> slice(name.span.start, name.span.length) => functionName
        "point|$(functionName)|$(point.state)|$(point.kind)|$(token.span.start)|$(location.line)|$(location.column)" -> println
    }
}
