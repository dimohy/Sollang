namespace sollang.compiler.llvm.text

import sollang.compiler.ir.typed as typedIr
import sollang.compiler.llvm.emitter.context as emitterContext
import syntax.generated.slg as grammar

isUserBlockScope node: typedIr.TypedIrNode -> Bool {
    node.kind == 19 and node.opcode == -227 and node.symbol != -2
}

functionNeedsStructuredBody functionIndex: Int, end: Int, context: ref emitterContext.EmitContext -> Bool {
    functionIndex + 1 => index!
    index! < end -> while {
        context.ir[index!] => node
        (node -> isUserBlockScope) or (node.kind == 6 and (node.opcode -> typedIr.isDirectIterationOpcode))
            or node.kind == 21 or node.kind == 22 -> if { true -> return }
        index! + 1 => index!
    }
    false
}

isUserBlockBorrowedRoleBinding index: Int, context: ref emitterContext.EmitContext -> Bool {
    context.ir[index] => binding
    binding.kind != 17 or binding.flags != 0 or binding.symbol < 0
        or binding.parent < 0 or binding.operand0 < 0
        -> if { false -> return }
    context.ir[binding.parent] => scope
    not (scope -> isUserBlockScope) or scope.symbol != -1
        -> if { false -> return }
    context.ir[binding.operand0].parent != binding.parent
        -> if { false -> return }
    context.ranges[binding.sourceModule] => range
    context.symbols[range.symbolStart + binding.symbol] => role
    role.kind == 35 and role.astNode == binding.astNode
}

# Original callback children retain provenance beneath a deferred region.
# Their copies execute in the caller's invocation and yield scopes.
userBlockCallbackOwner nodeIndex: Int, context: ref emitterContext.EmitContext -> Int {
    nodeIndex => ancestor!
    ancestor! >= 0 -> while {
        context.ir[ancestor!].parent => parent
        parent < 0 -> if { -1 -> return }
        context.ir[parent] => candidate
        candidate.kind == 19 and candidate.opcode == -227 and candidate.symbol == -2
            -> if { parent -> return }
        candidate.kind == 6 and candidate.targetModule >= 0 and candidate.symbol >= 0
            and (context -> emitterContext.genericFunctionForCall(parent)) >= 0
            -> if {
                context.ranges[candidate.targetModule] => range
                context.symbols[range.symbolStart + candidate.symbol] => declaration
                declaration.blockTypeNode >= 0 and declaration.flags / 64 % 2 == 0
                    and ancestor! != candidate.operand0 -> if { parent -> return }
            }
        parent => ancestor!
    }
    -1
}

# Runtime callback effects execute at their producer yield, while the original
# AST remains authoritative for lexical names and diagnostics.
schedulingSourceStart node: typedIr.TypedIrNode, context: ref emitterContext.EmitContext, state: ref CoreEmitterState -> UIntSize {
    node.parent => ancestor!
    ancestor! >= 0 -> while {
        context.ir[ancestor!] => owner
        owner.opcode == -227 and (owner.kind == 9 or owner.kind == 5 or owner.kind == 19)
            -> if { owner -> sourceStart(context, state) -> return }
        owner.parent => ancestor!
    }
    node -> sourceStart(context, state)
}

# Preserve lexical statement order inside one yield, as well as producer order
# between yields. Equal anchors must not erase callback binding dependencies.
schedulingPrecedes first: typedIr.TypedIrNode, second: typedIr.TypedIrNode, context: ref emitterContext.EmitContext, state: ref CoreEmitterState -> Bool {
    first -> schedulingSourceStart(context, state) => firstStart
    second -> schedulingSourceStart(context, state) => secondStart
    firstStart != secondStart -> if { firstStart < secondStart -> return }
    (first -> sourceStart(context, state)) < (second -> sourceStart(context, state))
}

# A suspension and its operand prerequisites belong to one source statement.
# Region boundaries keep a nested statement distinct from its enclosing control
# expression; function/body regions are never statement owners.
schedulingStatementOwner nodeIndex: Int, functionIndex: Int, context: ref emitterContext.EmitContext -> Int {
    nodeIndex => owner!
    context.ir[owner!].parent => parent!
    parent! > functionIndex -> while {
        context.ir[parent!].kind => parentKind
        parentKind == 1 or parentKind == 11 or parentKind == 19 or parentKind == 20
            -> if { owner! -> return }
        parent! => owner!
        context.ir[owner!].parent => parent!
    }
    owner!
}

parallelCallbackOwner nodeIndex: Int, context: ref emitterContext.EmitContext -> Int {
    -1 => owner!
    nodeIndex => callbackAncestor!
    callbackAncestor! >= 0 and owner! < 0 -> while {
        context.ir[callbackAncestor!].parent => callbackParent
        callbackParent >= 0 -> if {
            context.ir[callbackParent] => callbackParentNode
            callbackParentNode.kind == 6
                and (callbackParentNode.opcode == -207 or callbackParentNode.opcode == -209)
                and callbackAncestor! != callbackParentNode.operand0
                -> if { callbackParent => owner! } else { callbackParent => callbackAncestor! }
        } else {
            -1 => callbackAncestor!
        }
    }
    owner!
}

# A role receiver always has the exact element type of its structured source.
# Late call-plan repair can prepend that canonical receiver while an explicit
# reference argument still carries the role symbol provisionally. The explicit
# argument remains in the final call chain, so liveness cannot distinguish it;
# exact source-element type identity can.
parallelRoleMatchesSourceType nodeIndex: Int, parallelIndex: Int, context: ref emitterContext.EmitContext -> Bool {
    false => matches!
    context.ir[parallelIndex] => parallel
    parallel.operand0 >= 0 and parallel.operand0 < (context.ir -> len)
        -> if {
            context.ir[parallel.operand0] => source
            source.typeId >= 0 and source.typeId < (context.types -> len)
                -> if {
                    source.typeId => sourceTypeId!
                    context.types[sourceTypeId!].kind == 8
                        -> if { context.types[sourceTypeId!].first => sourceTypeId! }
                    sourceTypeId! >= 0
                        and sourceTypeId! < (context.types -> len)
                        and (context.types[sourceTypeId!].kind == 3
                        or context.types[sourceTypeId!].kind == 4
                        or context.types[sourceTypeId!].kind == 12)
                        -> if {
                            context.ir[nodeIndex].typeId == context.types[sourceTypeId!].first
                                => matches!
                        }
                }
        }
    matches!
}

parallelRoleSymbolOwner sourceModule: Int, symbolIndex: Int, beforeIndex: Int, context: ref emitterContext.EmitContext -> Int {
    sourceModule < 0 or sourceModule >= (context.ranges -> len)
        -> if { -1 -> return }
    context.ranges[sourceModule] => range
    symbolIndex < 0 or symbolIndex >= range.symbolCount
        -> if { -1 -> return }
    context.symbols[range.symbolStart + symbolIndex] => symbol
    symbol.kind != 35 or beforeIndex < 0 -> if { -1 -> return }
    beforeIndex -> parallelCallbackOwner(context) => owner!
    owner! >= 0 -> while {
        context.ir[owner!] => candidate
        candidate.sourceModule == sourceModule and candidate.astNode == symbol.astNode
            -> if { owner! -> return }
        owner! -> parallelCallbackOwner(context) => owner!
    }
    -1
}

structuredParallelRoleOwner nodeIndex: Int, context: ref emitterContext.EmitContext -> Int {
    -1 => owner!
    context.ir[nodeIndex] => role
    role.kind == 5
        and role.symbol >= 0
        and role.sourceModule >= 0
        and role.sourceModule < (context.ranges -> len)
        and role.symbol < context.ranges[role.sourceModule].symbolCount
        -> if {
            context.symbols[context.ranges[role.sourceModule].symbolStart + role.symbol] => roleSymbol
            roleSymbol.kind == 35 -> if {
                role.sourceModule -> parallelRoleSymbolOwner(role.symbol, nodeIndex, context) => owner!
                owner! >= 0 and not (nodeIndex -> parallelRoleMatchesSourceType(owner!, context))
                    -> if { -1 => owner! }
            }
        }
    owner!
}

emitStructuredParallelRoleValues parallelIndex: Int, context: ref emitterContext.EmitContext, state: ref CoreEmitterState -> Unit uses Console {
    0 => roleIndex!
    roleIndex! < (context.ir -> len) -> while {
        context.ir[roleIndex!] => role
        (roleIndex! -> structuredParallelRoleOwner(context)) == parallelIndex
            -> if {
                "  %v$(roleIndex!) = freeze " -> print
                role -> writeIrType(context, state)
                " %v$(parallelIndex)_item" -> println
            }
        roleIndex! + 1 => roleIndex!
    }
}

callHasDeclaredEffects node: typedIr.TypedIrNode, context: ref emitterContext.EmitContext, state: ref CoreEmitterState -> Bool {
    false => hasEffects!
    node.kind == 6 and node.targetModule >= 0 and node.targetModule < (context.ranges -> len) and node.symbol >= 0
        -> if {
            context.ranges[node.targetModule].symbolStart + node.symbol => targetGlobalSymbol
            targetGlobalSymbol >= 0 and targetGlobalSymbol < (state.frozenFunctionsHaveDeclaredEffectsBySymbol -> len)
                -> if { state.frozenFunctionsHaveDeclaredEffectsBySymbol[targetGlobalSymbol] => hasEffects! }
        }
    hasEffects!
}
# Mutation barriers share one catalog across sparse discovery, mutable reads,
# effect roots and predecessor ordering. In particular, reserve may replace
# the backing allocation even when the logical array length does not change.
# Fold owns a runtime body just like the other ordered controls. Its body may
# read or mutate earlier lexical storage before the fold publishes a result.
isSchedulingControlEffect node: typedIr.TypedIrNode -> Bool => node.kind == 18 or node.kind == 20 or node.kind == 27 or node.kind == 32 or node.kind == 34

# Opening a loop defers outside statements until its body is scheduled. Commit
# preceding bindings and completed control-result wrappers before that boundary;
# otherwise a body control can wait on a wrapper stranded outside its own loop.
needsIterationPredecessorCommit nodeIndex: Int, context: ref emitterContext.EmitContext -> Bool {
    context.ir[nodeIndex] => node
    node.kind == 17 -> if {
        not (node.operand0 >= 0 and context.ir[node.operand0].kind == 40) -> return
    }
    node.kind == 9 and node.opcode == -1
        and node.operand1 >= 0 and node.operand1 < (context.ir -> len)
        -> if {
            context.ir[node.operand1] => control
            (control.parent == nodeIndex and ((control -> isSchedulingControlEffect) or control.kind == 33)) -> return
        }
    false
}

isSchedulingMutationIntrinsic node: typedIr.TypedIrNode, context: ref emitterContext.EmitContext -> Bool {
    (node -> isResolvedAwaitNode(context))
        or ((node.kind == 6 or node.kind == 9) and (node.opcode -> typedIr.isFileRuntimeOpcode))
        or node.kind == 44 or (node.kind == 9 and (node.opcode == -204 or node.opcode == -213
        or node.opcode == -223 or node.opcode == -303
        or node.opcode == -229 or node.opcode == -241 or node.opcode == -242
        or node.opcode == -246 or node.opcode == -247 or node.opcode == -250
        or node.opcode == -251 or node.opcode == -252 or node.opcode == -254
        or node.opcode == -255 or node.opcode == -257 or node.opcode == -271
        or node.opcode == -306 or node.opcode == -307
        or node.opcode == -321))
        or (node -> typedIr.isArrayExchangeIntrinsicCall)
        or (node -> typedIr.isOwnedContainerTransform)
        or ((node.kind == 6 or node.kind == 9) and (node.opcode -> typedIr.isRawArenaMutationOpcode))
}

# Aggregate materialization captures inline values from backing storage.
# Later mutation must not overtake that capture after its descriptor read.
isSchedulingComparisonTextProducer node: typedIr.TypedIrNode, context: ref emitterContext.EmitContext, state: ref CoreEmitterState -> Bool {
    node.kind == 2 and node.parent >= 0 -> unless { false -> return }
    context.ir[node.parent] => consumer
    consumer.kind == 8 and (consumer.opcode == grammar.tokenIdEqualEqual() or consumer.opcode == grammar.tokenIdBangEqual())
        -> unless { false -> return }
    node -> sourceInterpolations(context, state) => interpolations
    interpolations -> each interpolationNode {
        interpolationNode.sourceToken == node.payloadToken and interpolationNode.parent < 0
            -> if { true -> return }
    }
    false
}

isSchedulingValueCapture node: typedIr.TypedIrNode, context: ref emitterContext.EmitContext, state: ref CoreEmitterState -> Bool {
    # A restored frame read belongs to its lexical continuation. Hoisting it
    # before an await creates an SSA value unavailable on the resume edge.
    node.kind == 5 and node.operand0 >= 0 -> if {
        node.parent => readOwner!
        readOwner! >= 0 and context.ir[readOwner!].kind != 0
            -> while { context.ir[readOwner!].parent => readOwner! }
        (node.operand0 -> asyncFrameBindingRoot(readOwner!, context, state)) >= 0
            -> if { true -> return }
    }
    node.kind == 17 and node.symbol >= 0 -> if {
        node.parent => owner!
        owner! >= 0 and context.ir[owner!].kind != 0 -> while { context.ir[owner!].parent => owner! }
        false => frameCapture!
        state.coroutine.slots -> each slot {
            slot.functionIr == owner! and slot.sourceModule == node.sourceModule and slot.symbol == node.symbol
                -> if { true => frameCapture! }
        }
        frameCapture! -> if { true -> return }
    }
    node.kind == 12 or node.kind == 14 or node.kind == 16 or node.kind == 35
        or ((node.kind == 6 or node.kind == 9) and (node.opcode == -342 or node.opcode == -344))
        or (node -> isSchedulingComparisonTextProducer(context, state))
}

# Deferred Text interpolation reads lexical bindings outside the Typed IR
# operand graph. Both entry and ordinary-function schedulers use this edge.
materializeLexicalBindingsReady materializeIndex: Int, ownerIndex: Int, expressionStart: Int, expressionEnd: Int, scheduled: ref [Bool; ~], context: ref emitterContext.EmitContext, state: ref CoreEmitterState -> Bool {
    context.ir[materializeIndex] => materialize
    -1 => textIndex!
    materialize.kind == 9 and materialize.opcode == -229
        and materialize.operand0 >= expressionStart and materialize.operand0 < expressionEnd
        -> if {
            materialize.operand0 -> producerValueIndex(context, state) => textIndex!
        } else {
            materialize -> isSchedulingComparisonTextProducer(context, state)
                -> if { materializeIndex => textIndex! }
        }
    textIndex! < 0 -> if { true -> return }
    context.ir[textIndex!] => textNode
    textNode -> sourceInterpolations(context, state) => interpolations
    0 => interpolationIndex!
    interpolationIndex! < (interpolations -> len)
        -> while {
            interpolations[interpolationIndex!] => interpolationNode
            interpolationNode.sourceToken == textNode.payloadToken and interpolationNode.kind == 1
                -> if {
                    interpolationNode -> interpolationBindingIndex(textNode.sourceModule, ownerIndex, materializeIndex, context, state) => bindingIndex
                    bindingIndex >= expressionStart and bindingIndex < expressionEnd and not scheduled[bindingIndex - expressionStart]
                        -> if { false -> return }
                }
            interpolationIndex! + 1 => interpolationIndex!
        }
    true
}

prepareFunctionExpressionOrder functionIndex: Int, functionEnd: Int, expressionStart: Int, context: ref emitterContext.EmitContext, state: ref CoreEmitterState -> [Int; ~] uses Console {
    # Mutable lexical roots require a function-local name and visibility scan.
    # Cache every binding/read root once because readiness and effect ordering
    # revisit the same nodes many times.
    [Int; ~] => functionMutableRoots!
    functionIndex + 1 => mutableRootCacheIndex!
    mutableRootCacheIndex! < functionEnd -> while {
        context.ir[mutableRootCacheIndex!] => mutableRootCacheNode
        -1 => mutableRootCacheValue!
        mutableRootCacheNode.kind == 5 or mutableRootCacheNode.kind == 10 or mutableRootCacheNode.kind == 17
            -> if { mutableRootCacheIndex! -> mutableBindingRoot(context, state) => mutableRootCacheValue! }
        functionMutableRoots! -> push(mutableRootCacheValue!)
        mutableRootCacheIndex! + 1 => mutableRootCacheIndex!
    }
    [Bool; ~] => expressionScheduled!
    [Bool; ~] => schedulingSuspensions!
    [Int; ~] => schedulingSuspensionIndices!
    [Int; ~] => schedulingSuspensionStatements!
    expressionStart => scheduledInit!
    scheduledInit! < functionEnd -> while {
        expressionScheduled! -> push(false)
        schedulingSuspensions! -> push(false)
        scheduledInit! + 1 => scheduledInit!
    }
    # A coroutine resume edge invalidates SSA reads from its earlier segment.
    # Use the frozen coroutine plan rather than inferring suspension by spelling.
    state.coroutine.points -> each suspensionPoint {
        suspensionPoint.functionIr == functionIndex
            and suspensionPoint.awaitIr >= expressionStart and suspensionPoint.awaitIr < functionEnd
            -> if {
                true => schedulingSuspensions![suspensionPoint.awaitIr - expressionStart]
                schedulingSuspensionIndices! -> push(suspensionPoint.awaitIr)
                schedulingSuspensionStatements! -> push(suspensionPoint.awaitIr -> schedulingStatementOwner(functionIndex, context))
            }
    }
    # Control and logical containment are structural properties of the frozen
    # Typed IR. Compute them once; readiness revisits must not repeat the same
    # ancestor and condition scans for every scheduling pass.
    [Bool; ~] => ifConditionRoots!
    expressionStart => ifConditionRootIndex!
    ifConditionRootIndex! < functionEnd -> while {
        ifConditionRoots! -> push(false)
        ifConditionRootIndex! + 1 => ifConditionRootIndex!
    }
    expressionStart => ifConditionIndex!
    ifConditionIndex! < functionEnd -> while {
        context.ir[ifConditionIndex!] => ifConditionCandidate
        ifConditionCandidate.kind == 18
            and ifConditionCandidate.operand0 >= expressionStart
            and ifConditionCandidate.operand0 < functionEnd
            -> if {
                context.ir[ifConditionCandidate.operand0] => ifConditionRoot
                (ifConditionRoot.kind == 8 and (ifConditionRoot.opcode == -25 or ifConditionRoot.opcode == -24))
                    or (ifConditionRoot.kind == 7 and ifConditionRoot.opcode == -26)
                    -> if { true => ifConditionRoots![ifConditionCandidate.operand0 - expressionStart] }
            }
        ifConditionIndex! + 1 => ifConditionIndex!
    }
    [Bool; ~] => expressionInsideControlRegion!
    [Bool; ~] => expressionInsideLogicalValue!
    [Bool; ~] => schedulingValueCaptures!
    expressionStart => containmentIndex!
    containmentIndex! < functionEnd -> while {
        context.ir[containmentIndex!] => containmentNode
        containmentNode.parent => containmentAncestor!
        containmentNode.kind == 19 or ifConditionRoots![containmentIndex! - expressionStart] => containmentInsideControl!
        containmentAncestor! >= expressionStart and not containmentInsideControl!
            -> while {
                context.ir[containmentAncestor!].kind == 19
                    or context.ir[containmentAncestor!].kind == 20
                    or context.ir[containmentAncestor!].kind == 33
                    or ifConditionRoots![containmentAncestor! - expressionStart]
                    -> if { true => containmentInsideControl! } else { context.ir[containmentAncestor!].parent => containmentAncestor! }
            }
        (containmentIndex! -> parallelCallbackOwner(context)) >= 0
            or (containmentIndex! -> userBlockCallbackOwner(context)) >= 0
            -> if { true => containmentInsideControl! }
        false => containmentInsideLogical!
        containmentNode.parent => containmentLogicalAncestor!
        containmentLogicalAncestor! >= expressionStart and not containmentInsideLogical!
            -> while {
                context.ir[containmentLogicalAncestor!] => containmentLogicalAncestorNode
                containmentLogicalAncestorNode.kind == 8 and (containmentLogicalAncestorNode.opcode == -24 or containmentLogicalAncestorNode.opcode == -25)
                    -> if { true => containmentInsideLogical! } else { containmentLogicalAncestorNode.parent => containmentLogicalAncestor! }
            }
        containmentNode.kind == 17 and containmentNode.operand0 >= expressionStart and containmentNode.operand0 < functionEnd and not containmentInsideControl!
            -> if {
                context.ir[containmentNode.operand0].parent => containmentBindingAncestor!
                containmentBindingAncestor! >= expressionStart and not containmentInsideControl!
                    -> while {
                        context.ir[containmentBindingAncestor!].kind == 19
                            -> if { true => containmentInsideControl! } else { context.ir[containmentBindingAncestor!].parent => containmentBindingAncestor! }
                    }
            }
        expressionInsideControlRegion! -> push(containmentInsideControl!)
        expressionInsideLogicalValue! -> push(containmentInsideLogical!)
        containmentNode -> isSchedulingValueCapture(context, state) => schedulingValueCapture
        schedulingValueCaptures! -> push(schedulingValueCapture)
        containmentIndex! + 1 => containmentIndex!
    }
    # Read and effect ordering only inspect nodes that can become observable
    # barriers. Freeze that sparse set once instead of rescanning every pure
    # expression on each readiness revisit.
    [Int; ~] => schedulingBarrierIndices!
    expressionStart => schedulingBarrierIndex!
    schedulingBarrierIndex! < functionEnd -> while {
        context.ir[schedulingBarrierIndex!] => schedulingBarrier
        schedulingSuspensions![schedulingBarrierIndex! - expressionStart]
            or schedulingValueCaptures![schedulingBarrierIndex! - expressionStart]
            or schedulingBarrier.kind == 6
            or (schedulingBarrier -> isSchedulingControlEffect)
            or schedulingBarrier.kind == 24
            or schedulingBarrier.kind == 25
            or (schedulingBarrier.kind == 17 and schedulingBarrier.flags == 1)
            or ((schedulingBarrier -> isSchedulingMutationIntrinsic(context)) or (schedulingBarrier.kind == 9 and (schedulingBarrier.opcode == -205 or schedulingBarrier.opcode == -220 or schedulingBarrier.opcode == -221 or schedulingBarrier.opcode == -304 or schedulingBarrier.opcode == -305 or schedulingBarrier.opcode == -253 or schedulingBarrier.opcode == -256)))
            -> if { schedulingBarrierIndices! -> push(schedulingBarrierIndex!) }
        schedulingBarrierIndex! + 1 => schedulingBarrierIndex!
    }
    [Int; ~] => expressionOrder!
    true => scheduleProgress!
    scheduleProgress! -> while {
        false => scheduleProgress!
        -1 => pendingEach!
        Int(expressionOrder! -> len) - 1 => pendingEachOrderIndex!
        pendingEachOrderIndex! >= 0 and pendingEach! < 0
            -> while {
                expressionOrder![pendingEachOrderIndex!] => orderedEachCandidate
                context.ir[orderedEachCandidate] => orderedEachNode
                orderedEachNode.kind == 6 and (orderedEachNode.opcode -> typedIr.isDirectIterationOpcode)
                    -> if {
                        false => orderedEachHasPendingBody!
                        expressionStart => orderedEachBodySearch!
                        orderedEachBodySearch! < functionEnd and not orderedEachHasPendingBody!
                            -> while {
                                NodeAncestorRequest { nodeIndex: orderedEachBodySearch!, ownerIndex: orderedEachCandidate } => orderedEachAncestorRequest
                                orderedEachAncestorRequest -> irDescendsFrom(context, state) => orderedEachInside
                                not expressionScheduled![orderedEachBodySearch! - expressionStart] and orderedEachInside
                                    -> if { true => orderedEachHasPendingBody! }
                                orderedEachBodySearch! + 1 => orderedEachBodySearch!
                            }
                        orderedEachHasPendingBody! -> if { orderedEachCandidate => pendingEach! }
                    }
                pendingEachOrderIndex! - 1 => pendingEachOrderIndex!
            }
        expressionStart => scheduleCandidate!
        scheduleCandidate! < functionEnd -> while {
            not expressionScheduled![scheduleCandidate! - expressionStart]
                -> if {
                    context.ir[scheduleCandidate!] => scheduleNode
                    expressionInsideControlRegion![scheduleCandidate! - expressionStart] => insideControlRegion
                    expressionInsideLogicalValue![scheduleCandidate! - expressionStart] => insideLogicalValue
                    insideControlRegion or scheduleNode.kind == 19
                        -> if {
                            true => expressionScheduled![scheduleCandidate! - expressionStart]
                            true => scheduleProgress!
                        }
                    true => scheduleReady!
                    insideControlRegion or scheduleNode.kind == 19
                        -> if { false => scheduleReady! }
                    scheduleReady! and scheduleNode.kind == 6 and (scheduleNode.opcode -> typedIr.isDirectIterationOpcode)
                        -> if {
                            expressionStart => predecessor!
                            predecessor! < scheduleCandidate! and scheduleReady!
                                -> while {
                                    context.ir[predecessor!].parent == scheduleNode.parent
                                        and (predecessor! -> needsIterationPredecessorCommit(context))
                                        and (context.ir[predecessor!] -> schedulingPrecedes(scheduleNode, context, state))
                                        and not expressionScheduled![predecessor! - expressionStart]
                                        -> if { false => scheduleReady! }
                                    predecessor! + 1 => predecessor!
                                }
                        }
                    false => scheduleInsidePendingEach!
                    pendingEach! >= 0 -> if {
                        NodeAncestorRequest { nodeIndex: scheduleCandidate!, ownerIndex: pendingEach! } => pendingEachAncestorRequest
                        pendingEachAncestorRequest -> irDescendsFrom(context, state) => scheduleInsidePendingEach!
                    }
                    scheduleReady! and pendingEach! >= 0 and not scheduleInsidePendingEach!
                        -> if { false => scheduleReady! }
                    scheduleReady! and scheduleNode.parent >= expressionStart
                        -> if {
                            scheduleCandidate! => eachBranchChild!
                            scheduleNode.parent => eachScheduleAncestor!
                            eachScheduleAncestor! >= expressionStart and scheduleReady!
                                -> while {
                                    context.ir[eachScheduleAncestor!] => eachScheduleOwner
                                    eachScheduleOwner.kind == 6 and (eachScheduleOwner.opcode -> typedIr.isDirectIterationOpcode) and eachBranchChild! != eachScheduleOwner.operand0 and not expressionScheduled![eachScheduleAncestor! - expressionStart]
                                        -> if { false => scheduleReady! }
                                    eachScheduleAncestor! => eachBranchChild!
                                    context.ir[eachScheduleAncestor!].parent => eachScheduleAncestor!
                                }
                        }
                    scheduleReady! and scheduleNode.kind != 38 and scheduleNode.operand0 >= expressionStart and scheduleNode.operand0 < functionEnd and (scheduleNode.kind != 5 or context.ir[scheduleNode.operand0].flags % 2 == 0) and not expressionScheduled![scheduleNode.operand0 - expressionStart]
                        -> if { false => scheduleReady! }
                    # Assignment operand0 may be a non-emitting value wrapper.
                    # Wait for the canonical child that the emitter reads, so a
                    # post-control assignment never references a later SSA value.
                    scheduleReady! and (scheduleNode.kind == 24 or scheduleNode.kind == 25) and scheduleNode.operand0 >= expressionStart
                        -> if {
                            scheduleNode.operand0 -> aggregateValueIndex(context, state) => scheduleAssignmentValue
                            scheduleAssignmentValue >= expressionStart
                                and scheduleAssignmentValue < functionEnd
                                and not expressionScheduled![scheduleAssignmentValue - expressionStart]
                                -> if { false => scheduleReady! }
                            expressionStart => scheduleAssignmentChildSearch!
                            scheduleAssignmentChildSearch! < functionEnd and scheduleReady!
                                -> while {
                                    context.ir[scheduleAssignmentChildSearch!].parent == scheduleNode.operand0
                                        and not expressionScheduled![scheduleAssignmentChildSearch! - expressionStart]
                                        -> if { false => scheduleReady! }
                                    scheduleAssignmentChildSearch! + 1 => scheduleAssignmentChildSearch!
                                }
                        }
                    scheduleReady! and scheduleNode.kind == 5 and scheduleNode.operand0 >= expressionStart and scheduleNode.operand0 < functionEnd and context.ir[scheduleNode.operand0].kind == 17 and context.ir[scheduleNode.operand0].flags == 1
                        -> if {
                            functionMutableRoots![scheduleNode.operand0 - functionIndex - 1] => scheduleMutableReadRoot
                            # A mutable rebind reads the slot value that existed
                            # before its own store. Do not turn that read into a
                            # dependency on the store currently being scheduled.
                            scheduleMutableReadRoot != scheduleNode.operand0 and scheduleMutableReadRoot >= expressionStart and scheduleMutableReadRoot < functionEnd and not expressionScheduled![scheduleMutableReadRoot - expressionStart]

                                            not expressionScheduled![earlierEffectSearch - expressionStart] and (schedulingSuspensions![earlierEffectSearch - expressionStart] or schedulingValueCaptures![earlierEffectSearch - expressionStart] or earlierEffect.kind == 6 or (earlierEffect -> isSchedulingControlEffect) or earlierEffect.kind == 23 or earlierEffect.kind == 24 or earlierEffect.kind == 25 or (earlierEffect.kind == 17 and earlierEffect.flags == 1) or (earlierEffect -> isSchedulingMutationIntrinsic(context))) and (earlierEffect -> schedulingPrecedes(context.ir[effectOrderingIndex!], context, state))
                                                -> if {
                                                    # A call producing an effect's source is its input,
                                                    # not a sibling ordered after that enclosing effect.
                                                    NodeAncestorRequest { nodeIndex: scheduleCandidate!, ownerIndex: earlierEffectSearch } -> irDescendsFrom(context, state) => insideEarlierEffect
                                                    earlierEffect.parent => earlierEffectAncestor!
                                                    true => earlierRootEffect!
                                                    false => earlierInsideRegion!
                                                    earlierEffectAncestor! >= expressionStart and earlierRootEffect! and not earlierInsideRegion!
                                                        -> while {
                                                            context.ir[earlierEffectAncestor!].kind == 19 or context.ir[earlierEffectAncestor!].kind == 20
                                                                or context.ir[earlierEffectAncestor!].kind == 38
                                                                -> if { true => earlierInsideRegion! } else {
                                                                    (schedulingValueCaptures![earlierEffectSearch - expressionStart] and schedulingValueCaptures![earlierEffectAncestor! - expressionStart]) or (context.ir[earlierEffectAncestor!].kind == 6 and not (context.ir[earlierEffectAncestor!].opcode -> typedIr.isDirectIterationOpcode)) or (context.ir[earlierEffectAncestor!] -> isSchedulingControlEffect) or context.ir[earlierEffectAncestor!].kind == 23 or context.ir[earlierEffectAncestor!].kind == 24 or context.ir[earlierEffectAncestor!].kind == 25 or (context.ir[earlierEffectAncestor!].kind == 17 and context.ir[earlierEffectAncestor!].flags == 1) or (context.ir[earlierEffectAncestor!].kind == 9 and (context.ir[earlierEffectAncestor!].opcode == -204 or context.ir[earlierEffectAncestor!].opcode == -205 or context.ir[earlierEffectAncestor!].opcode == -213 or context.ir[earlierEffectAncestor!].opcode == -220 or context.ir[earlierEffectAncestor!].opcode == -221 or context.ir[earlierEffectAncestor!].opcode == -304 or context.ir[earlierEffectAncestor!].opcode == -305 or context.ir[earlierEffectAncestor!].opcode == -229 or context.ir[earlierEffectAncestor!].opcode == -241 or context.ir[earlierEffectAncestor!].opcode == -242 or context.ir[earlierEffectAncestor!].opcode == -246 or context.ir[earlierEffectAncestor!].opcode == -247 or context.ir[earlierEffectAncestor!].opcode == -250 or context.ir[earlierEffectAncestor!].opcode == -251 or context.ir[earlierEffectAncestor!].opcode == -252 or context.ir[earlierEffectAncestor!].opcode == -254 or context.ir[earlierEffectAncestor!].opcode == -255 or context.ir[earlierEffectAncestor!].opcode == -257 or (context.ir[earlierEffectAncestor!].opcode -> typedIr.isRawArenaMutationOpcode)))
                                                                        -> if { false => earlierRootEffect! } else { context.ir[earlierEffectAncestor!].parent => earlierEffectAncestor! }
                                                                }
                                                        }
                                                    earlierRootEffect! and not earlierInsideRegion! and not insideEarlierEffect
                                                        -> if { false => scheduleReady! }
                                                }
                                            earlierEffectCursor! + 1 => earlierEffectCursor!
                                        }
                                }
                        }
                    # An immutable alias can relax data dependencies, but must
                    # not escape an active each before its remaining body work.
                    not insideControlRegion
                        and (pendingEach! < 0 or scheduleInsidePendingEach!)
                        and scheduleNode.kind == 5
                        and not schedulingValueCaptures![scheduleCandidate! - expressionStart]
                        and scheduleNode.operand0 >= expressionStart
                        and scheduleNode.operand0 < functionEnd
                        and context.ir[scheduleNode.operand0].kind == 17
                        and context.ir[scheduleNode.operand0].flags != 1
                        and scheduleNode.parent >= expressionStart
                        and scheduleNode.parent < functionEnd
                        and (context.ir[scheduleNode.parent].kind == 12
                        or context.ir[scheduleNode.parent].kind == 14
                        or context.ir[scheduleNode.parent].kind == 16
                        or context.ir[scheduleNode.parent].kind == 26
                        or context.ir[scheduleNode.parent].kind == 35)
                        and expressionScheduled![scheduleNode.operand0 - expressionStart]
                        -> if {
                            true => scheduleReady!
                        }
                    # Pure SSA values cannot be hoisted across an earlier
                    # suspension statement unless their storage is captured.
                    # Keep the reads in their source segment instead of adding
                    # accidental spills for values declared after suspension.
                    scheduleReady! and (schedulingSuspensionIndices! -> len) > 0
                        -> if {
                            scheduleCandidate! -> schedulingStatementOwner(functionIndex, context) => statementOwner
                            context.ir[statementOwner] => statement
                            0 => suspensionCursor!
                            scheduleReady! and suspensionCursor! < (schedulingSuspensionIndices! -> len)
                                -> while {
                                    schedulingSuspensionIndices![suspensionCursor!] => suspensionIndex
                                    schedulingSuspensionStatements![suspensionCursor!] => suspensionOwner
                                    context.ir[suspensionOwner] => suspensionStatement
                                    not expressionScheduled![suspensionIndex - expressionStart]
                                        and suspensionOwner != statementOwner
                                        and statement.sourceModule == suspensionStatement.sourceModule
                                        and statement.sourceModule >= 0
                                        and statement.astNode >= 0 and suspensionStatement.astNode >= 0
                                        and (suspensionStatement -> schedulingPrecedes(statement, context, state))
                                        -> if { false => scheduleReady! }
                                    suspensionCursor! + 1 => suspensionCursor!
                                }
                        }
                    scheduleReady! -> if {
                        not insideLogicalValue -> if { expressionOrder! -> push(scheduleCandidate!) }
                        true => expressionScheduled![scheduleCandidate! - expressionStart]
                        not insideLogicalValue
                            and scheduleNode.parent >= expressionStart
                            and scheduleNode.parent < functionEnd
                            and not expressionScheduled![scheduleNode.parent - expressionStart]
                            and context.ir[scheduleNode.parent].kind == 30
                            and context.ir[scheduleNode.parent].operand0 == scheduleCandidate!
                            and context.nodes[context.ranges[context.ir[scheduleNode.parent].sourceModule].astStart + context.ir[scheduleNode.parent].astNode].kind == 82
                            -> if {
                                expressionOrder! -> push(scheduleNode.parent)
                                true => expressionScheduled![scheduleNode.parent - expressionStart]
                            }
                        scheduleNode.kind == 6 and (scheduleNode.opcode -> typedIr.isDirectIterationOpcode)
                            -> if { scheduleCandidate! => pendingEach! }
                        true => scheduleProgress!
                    }
                }
            scheduleCandidate! + 1 => scheduleCandidate!
        }
    }
    # Once a top-level control returns on every path and is the final child of
    # its owning region, later scheduled expressions are unreachable. Use the
    # same wrapper-aware, later-sibling-aware predicate as control emission and
    # function return selection; a source fallback after the control requires
    # the control merge to remain materialized even if every arm returns.
    [Int; ~] => reachableExpressionOrder!
    true => reachableExpressionOpen!
    0 => reachableExpressionIndex!
    reachableExpressionIndex! < (expressionOrder! -> len)
        -> while {
            expressionOrder![reachableExpressionIndex!] => reachableCandidate
            reachableExpressionOpen! -> if {
                reachableExpressionOrder! -> push(reachableCandidate)
                context.ir[reachableCandidate].parent == context.ir[functionIndex].operand0
                    and (reachableCandidate -> terminatingControlReturns(context, state))
                    -> if { false => reachableExpressionOpen! }
            }
            reachableExpressionIndex! + 1 => reachableExpressionIndex!
        }
    reachableExpressionOrder!
}
