contextualizeNumericCallArgumentArena prepared: ref semanticContext.SemanticSnapshot, types: ref [typeIds.SemanticType; ~], expressions: mut [ExpressionTypeId; ~], expressionByAst: mut [Int; ~], sourceModule: Int, argumentAst: Int, expectedTypeId: Int -> Bool {
    expectedTypeId < 0 or expectedTypeId >= (types -> len)
        -> if { false -> return }
    types[expectedTypeId] => expected
    not (expected.kind == 1 and expected.origin == 1
        and ((expected.symbol >= 2 and expected.symbol <= 14)
        or (expected.symbol >= 19 and expected.symbol <= 22)))
        -> if { false -> return }
    prepared.package.ranges[sourceModule] => range
    range.astStart + argumentAst => globalAst
    prepared.package.nodes[globalAst] => argument
    false => contextual!
    argument.kind == 14 -> if {
        prepared.package.sources[sourceModule] -> len => sourceLength
        prepared.package.sources[sourceModule] -> slice(UIntSize(0), sourceLength) => source
        source -> numericLiteralRange.literalUsesFloatingSyntax(argument.start, argument.start + argument.length) => floating
        not floating or expected.symbol >= 19 -> if { true => contextual! }
    }
    argument.kind == 22 -> if {
        0 => childAst!
        childAst! < range.astCount -> while {
            prepared.package.nodes[range.astStart + childAst!] => child
            child.parent == argumentAst and child.kind == 14
                -> if {
                    contextualizeNumericCallArgumentArena(prepared, types, expressions, expressionByAst, sourceModule, childAst!, expectedTypeId) => childContextual
                    childContextual -> if { true => contextual! }
                }
            childAst! + 1 => childAst!
        }
    }
    argument.kind == 10 -> if {
        false => foundBranch!
        true => allBranchesContextual!
        0 => controlAst!
        controlAst! < range.astCount -> while {
            prepared.package.nodes[range.astStart + controlAst!] => control
            control.parent == argumentAst and control.kind == 42
                -> if {
                    0 => branchAst!
                    branchAst! < range.astCount -> while {
                        prepared.package.nodes[range.astStart + branchAst!] => branch
                        branch.parent == controlAst! and branch.kind == 43
                            -> if {
                                true => foundBranch!
                                -1 => valueAst!
                                0 => valueSearch!
                                valueSearch! < range.astCount -> while {
                                    prepared.package.nodes[range.astStart + valueSearch!] => value
                                    value.parent == branchAst!
                                        and value.start + value.length == branch.start + branch.length
                                        -> if { valueSearch! => valueAst! }
                                    valueSearch! + 1 => valueSearch!
                                }
                                valueAst! < 0 -> if { false => allBranchesContextual! } else {
                                    contextualizeNumericCallArgumentArena(prepared, types, expressions, expressionByAst, sourceModule, valueAst!, expectedTypeId) => branchContextual
                                    not branchContextual -> if { false => allBranchesContextual! }
                                }
                            }
                        branchAst! + 1 => branchAst!
                    }
                }
            controlAst! + 1 => controlAst!
        }
        foundBranch! and allBranchesContextual! -> if { true => contextual! }
    }
    (argument.kind == 20 or argument.kind == 21)
        and expected.symbol >= 2 and expected.symbol <= 14
        -> if {
            -1 => leftAst!
            -1 => rightAst!
            0 => childAst!
            childAst! < range.astCount -> while {
                prepared.package.nodes[range.astStart + childAst!].parent == argumentAst
                    -> if {
                        leftAst! < 0 -> if { childAst! => leftAst! } else { childAst! => rightAst! }
                    }
                childAst! + 1 => childAst!
            }
            leftAst! >= 0 and rightAst! >= 0 -> if {
                contextualizeNumericCallArgumentArena(prepared, types, expressions, expressionByAst, sourceModule, leftAst!, expectedTypeId) => leftContextual
                contextualizeNumericCallArgumentArena(prepared, types, expressions, expressionByAst, sourceModule, rightAst!, expectedTypeId) => rightContextual
                expressionByAst[range.astStart + leftAst!] => leftIndex
                expressionByAst[range.astStart + rightAst!] => rightIndex
                leftIndex >= 0 and rightIndex >= 0 -> if {
                    types -> typeIds.materializedReadTypeId(expressions[leftIndex].typeId) => leftType
                    types -> typeIds.materializedReadTypeId(expressions[rightIndex].typeId) => rightType
                    leftType == expectedTypeId and rightType == expectedTypeId
                        -> if { true => contextual! }
                }
            }
        }
    contextual! -> if {
        expressionByAst[globalAst] => expressionIndex
        expressionIndex >= 0 -> if {
            expressions[expressionIndex] => expression!
            expectedTypeId => expression!.typeId
            0 => expression!.status
            expressions -> replaceCallExpression(expressionIndex, expression!)
        } else {
            expressions -> len => nextIndex
            expressions -> push(ExpressionTypeId {
                sourceModule: sourceModule
                astNode: argumentAst
                typeId: expectedTypeId
                status: 0
            })
            nextIndex => expressionByAst[globalAst]
        }
    }
    contextual!
}
